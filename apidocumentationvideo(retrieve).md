    get:
      description: This endpoint is used to query the result of an asynchronous request.
      parameters:
        - $ref: '#/components/parameters/AcceptLanguage'

                  request_id:
                    description: >-

            application/json:
              schema:
                $ref: '#/components/schemas/Error'
